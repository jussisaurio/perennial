import { expect, test, type Page } from "@playwright/test";
import { appendFileSync, mkdirSync } from "node:fs";

const OUT = process.env.BT_OUT ?? "";
const WANT = (process.env.BT_CASE ?? "").split(",").filter((c) => c !== "");
if (WANT.length === 0 || OUT === "") throw new Error("set BT_CASE and BT_OUT");

const DAY = new Date("2026-10-11T12:00:00");
const NIGHT = new Date("2026-09-26T22:00:00");
const VIEWS = { desktop: { width: 1280, height: 720 }, phone: { width: 390, height: 844 } } as const;
type View = keyof typeof VIEWS;

type G = { x: number; z: number; yaw: number; pitch: number; elapsed?: number; layout: { gate: { x: number }; bounds: { maxZ: number } } };

/** One test's frames: `<n>-<prefix><view>.png` in the case's folder. */
type Shot = (n: number | string, what: string) => Promise<void>;

const frames = (page: Page) => page.evaluate(() => new Promise((done) => requestAnimationFrame(() => requestAnimationFrame(done))));

/** Wait in game time where the scene has a clock, or else on the wall clock. */
async function gameSeconds(page: Page, s: number) {
  const from = await page.evaluate(() => (window as unknown as { __garden: G }).__garden.elapsed);
  if (typeof from !== "number") return page.waitForTimeout(s * 1000);
  await page.waitForFunction((until) => ((window as unknown as { __garden: G }).__garden.elapsed ?? 0) >= until, from + s, { timeout: 60_000 });
}

async function hold(page: Page, key: string, s: number) {
  await page.keyboard.down(key);
  await gameSeconds(page, s);
  await page.keyboard.up(key);
  await frames(page);
}

async function openWalk(page: Page, words: number, at = DAY) {
  await page.clock.setFixedTime(at);
  await page.addInitScript(() => {
    (window as unknown as { __gardenDetail: string }).__gardenDetail = "normal";
  });
  await page.goto(`/?demo=es:${words}#/today`);
  const walk = page.getByRole("button", { name: "Walk in your garden" });
  await expect(walk).toBeVisible({ timeout: 120_000 });
  await walk.click();
  await expect(page.getByText(/^\d+ plants · \d+ in bloom/)).toBeVisible({ timeout: 60_000 });
  await page.waitForFunction(() => "__garden" in window);
  await expect(page.locator(".gw-loading")).toHaveCount(0, { timeout: 60_000 });
  await frames(page);
}

/** At the stile's foot, or `back` metres before it, facing the back fence. */
const atStile = (page: Page, back: number, pitch: number) =>
  page.evaluate(
    async ([back, pitch]) => {
      const { stileAt } = await import(/* @vite-ignore */ "/src/garden/stile.ts");
      const s = (window as unknown as { __garden: G }).__garden;
      const at = stileAt(s.layout);
      Object.assign(s, { x: at.foot[0], z: at.foot[1] + 0.2 + back, yaw: 0, pitch });
    },
    [back, pitch],
  );

const button = (page: Page, name: string) => page.getByRole("button", { name, exact: true });
const has = async (page: Page, name: string) => (await button(page, name).count()) > 0 && (await button(page, name).first().isVisible());

type Screen = "today" | "card" | "answer" | "words" | "settings";

/** Open one app screen of a fresh 30-word demo learner. */
async function openScreen(page: Page, screen: Screen) {
  await page.clock.setFixedTime(DAY);
  if (screen === "words" || screen === "settings") {
    const name = screen === "words" ? "Words" : "Settings";
    await page.goto(`/?demo=es:30#/${screen}`);
    await expect(page.getByRole("tab", { name })).toBeVisible({ timeout: 60_000 });
    await page.waitForTimeout(1500);
    return;
  }
  await page.goto("/?demo=es:30#/today");
  const start = page.getByRole("button", { name: /today's practice/i });
  await expect(start).toBeVisible({ timeout: 120_000 });
  await page.waitForTimeout(1500);
  if (screen === "today") return;
  await start.click();
  const card = page.locator("main.card");
  await expect(card).toBeVisible({ timeout: 30_000 });
  await page.waitForTimeout(800);
  if (screen === "card") return;
  // A writing card is checked with a wrong word; any other card is shown with Space.
  const field = page.getByRole("textbox");
  if (await field.isVisible()) {
    await field.fill("x");
    await page.keyboard.press("Enter");
  } else
    await expect(async () => {
      await page.keyboard.press("Space");
      await expect(card.locator('[aria-keyshortcuts="1"]')).toBeVisible({ timeout: 1000 });
    }).toPass();
  await page.waitForTimeout(800);
}

const SCREENS: Record<Screen, string> = {
  today: "the Today screen of a learner with 30 words",
  card: "the first card of a practice session",
  answer: "the first card of a practice session, after its answer is shown",
  words: "the Words screen",
  settings: "the Settings screen",
};

async function appScreens(page: Page, shoot: Shot) {
  for (const screen of Object.keys(SCREENS) as Screen[]) {
    await openScreen(page, screen);
    await shoot(screen, SCREENS[screen]);
  }
}

/**
 * Deliberate regressions on today's app (#1004), so a pairwise judge cannot win by
 * preferring the newer style. Each is one small CSS or text change on one screen.
 */
const PLANTS: { name: string; screen: Screen; what: string; css?: string; js?: string }[] = [
  { name: "misalign", screen: "today", what: "the date line moved 10 px right of the heading", css: ".home-head .date { margin-left: 10px !important; }" },
  { name: "contrast", screen: "today", what: "body text in a pale grey", css: ".page p, .muted { color: #c9c4ba !important; }" },
  { name: "cramped", screen: "settings", what: "spacing between sections and inside panels cut to 2 px", css: ".page > *, .panel, .form, section { margin-top: 2px !important; margin-bottom: 2px !important; padding-top: 2px !important; padding-bottom: 2px !important; gap: 2px !important; }" },
  { name: "jokey", screen: "today", what: "a jokey main button", js: "const w = document.createTreeWalker(document.body, NodeFilter.SHOW_TEXT); for (let n = w.nextNode(); n; n = w.nextNode()) if (/today's practice/i.test(n.textContent)) n.textContent = 'Feed the word-monster!';" },
  { name: "unequal", screen: "answer", what: "the two buttons under the card made different shapes and sizes", css: "main.card .actions > :first-child { border-radius: 4px !important; padding: 0.4rem 0.7rem !important; font-size: 0.85rem !important; align-self: center !important; }" },
  { name: "font", screen: "card", what: "the card's text in a different font", css: "main.card, main.card * { font-family: 'Times New Roman', serif !important; }" },
  { name: "colouronly", screen: "settings", what: "the chosen tab shown only by a green text colour", css: ".tab.on { background: transparent !important; box-shadow: none !important; color: #4f8a52 !important; }" },
  { name: "overlap", screen: "words", what: "a row's text drawn over the next row", css: ".word-row { margin-bottom: -14px !important; }" },
];

/** Each plant's screen as it is, then with the plant. */
async function planted(page: Page, shoot: Shot) {
  for (const p of PLANTS) {
    await openScreen(page, p.screen);
    await shoot(`${p.name}-base`, SCREENS[p.screen]);
    await page.evaluate(
      ([css, js]) => {
        const style = document.createElement("style");
        style.textContent = css;
        document.head.append(style);
        if (js !== "") new Function(js)();
      },
      [p.css ?? "", p.js ?? ""],
    );
    await page.waitForTimeout(300);
    await shoot(`${p.name}-bad`, `${SCREENS[p.screen]}; planted: ${p.what}`);
  }
}

test.describe.configure({ timeout: 300_000, mode: "serial" });
for (const id of WANT) {
  const c = CASES[id];
  if (c === undefined) throw new Error(`no case ${id}`);
  const dir = `${OUT}/${id}`;
  mkdirSync(dir, { recursive: true });
  for (const theme of c.themes ?? ["light"])
    for (const view of Object.keys(VIEWS) as View[])
      test(`${id} ${view} ${theme}`, async ({ page }) => {
        await page.emulateMedia({ colorScheme: theme });
        await page.setViewportSize(VIEWS[view]);
        // Screens shot in both themes get the theme in the file name.
        const suffix = `${c.themes === undefined ? "" : `${theme}-`}${view}`;
        await c.run(page, async (n, what) => {
          const name = `${n}-${suffix}.png`;
          await page.screenshot({ path: `${dir}/${name}` });
          appendFileSync(`${dir}/steps.txt`, `${name}: ${what}\n`);
        });
      });
}
